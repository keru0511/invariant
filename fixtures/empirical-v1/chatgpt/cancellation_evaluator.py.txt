class _InputError(Exception):
    def __init__(self, message, path, code="INVALID_ARGUMENT"):
        super().__init__(message)
        self.path = path
        self.code = code


def _fail(message, path, code="INVALID_ARGUMENT"):
    raise _InputError(message, path, code)


def _record(raw, allowed, path):
    # Exact built-ins avoid accepting bool as int or invoking subclass hooks.
    if type(raw) is not dict:
        _fail("Expected a plain JSON object.", path)
    if any(type(key) is not str or key not in allowed for key in raw):
        _fail("Unknown field.", path)
    return raw.copy()


def _request(raw, evaluation):
    allowed = ("domain", "version", "function", "args") if evaluation else ("domain", "version", "function")
    request = _record(raw, allowed, "$")
    if type(request.get("domain")) is not str or request["domain"] != "synthetic-cancellation":
        _fail("Expected domain synthetic-cancellation.", "$.domain", "UNKNOWN_DOMAIN")
    if type(request.get("version")) is not str or request["version"] != "cancellation-v1":
        _fail("Explicit version cancellation-v1 is required; no fallback version is selected.", "$.version", "VERSION_MISMATCH")
    if evaluation or "function" in request:
        if type(request.get("function")) is not str or request["function"] != "cancellation-fee":
            _fail("Expected function cancellation-fee.", "$.function", "UNKNOWN_FUNCTION")
    return request


def _boundary(action):
    try:
        return action()
    except _InputError as error:
        return {"status": "error", "error": {"code": error.code, "message": str(error), "path": error.path}}
    except Exception:
        return {"status": "error", "error": {"code": "INVALID_ARGUMENT", "message": "Input could not be read as plain JSON.", "path": "$"}}


def _valid_seconds(value):
    # JavaScript JSON numbers have one numeric type. Integral finite floats
    # therefore have the same meaning here, but Python bool is never a number.
    if type(value) not in (int, float):
        return False
    if type(value) is float:
        if not math.isfinite(value) or not value.is_integer():
            return False
        if value == 0 and math.copysign(1.0, value) < 0:
            return False
    return 0 <= value <= 9007199254740991


def _evaluate(raw):
    request = _request(raw, True)
    facts = _record(request.get("args"), ("organizerCancelled", "secondsUntilStart", "bookingPriceYen"), "$.args")
    if "organizerCancelled" in facts and type(facts["organizerCancelled"]) is not bool:
        _fail("Expected boolean, or omit an unknown fact.", "$.args.organizerCancelled")
    if "secondsUntilStart" in facts and not _valid_seconds(facts["secondsUntilStart"]):
        _fail("Expected nonnegative safe integer seconds before start (no negative zero).", "$.args.secondsUntilStart")
    if "bookingPriceYen" in facts:
        price = facts["bookingPriceYen"]
        if type(price) is not str or len(price) > 100 or re.fullmatch(r"[1-9][0-9]*", price) is None:
            _fail("Expected a canonical positive integer yen string of at most 100 digits.", "$.args.bookingPriceYen")

    # All supplied fields have now been validated, including irrelevant ones.
    def answer(decision, value, reason):
        return {**_target(), "status": "ok", "decision": decision, "abstain": False, "value": value, "reason": reason}

    def ask(field):
        return {**_target(), "status": "needs_information", "decision": None, "abstain": True, "missingFacts": [field]}

    if facts.get("organizerCancelled") is True:
        return answer("free", "0", "organizer_exception")
    if "secondsUntilStart" in facts and facts["secondsUntilStart"] >= 172800:
        return answer("free", "0", "at_least_48_hours")
    if "organizerCancelled" not in facts:
        return ask("organizerCancelled")
    if "secondsUntilStart" not in facts:
        return ask("secondsUntilStart")
    if "bookingPriceYen" not in facts:
        return ask("bookingPriceYen")
    yen = int(facts["bookingPriceYen"])
    if facts["secondsUntilStart"] >= 86400:
        return answer("half", str(yen // 2), "at_least_24_below_48_hours_floor_yen")
    return answer("full", str(yen), "below_24_hours")


def evaluate(request_dict):
    """Evaluate explicit pinned target and supplied facts without modifying input."""
    return _boundary(lambda: _evaluate(request_dict))

        _request(raw, False)
        tools = _tools()
        return {**_target(), "status": "ok", "scope": "synthetic_experiment_only", "ruleText": RULE_TEXT,
